# Build the portable C engine with MSYS2 separately; the GUI remains on its MSVC ABI.
set(AB_COREDEVICE_RUNTIME "${PROJECT_SOURCE_DIR}/build/coredevice-runtime" CACHE PATH "Prepared CoreDevice host/DLL/DDI directory")
if(EXISTS "${AB_COREDEVICE_RUNTIME}/AudioBridge.CoreDeviceHost.exe")
  add_custom_target(CoreDevice.Runtime ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${AB_COREDEVICE_RUNTIME}" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/CoreDevice")
  add_test(NAME CoreDevice.Runtime COMMAND "${AB_COREDEVICE_RUNTIME}/AudioBridge.CoreDeviceHost.exe" --check-runtime)
else()
  message(STATUS "CoreDevice runtime not prepared; run scripts/Build-CoreDeviceHost.ps1 to enable live capture")
endif()
add_executable(CoreDevice.TestHost "${PROJECT_SOURCE_DIR}/tests/CoreDeviceTestHost.cpp")
target_include_directories(CoreDevice.TestHost PRIVATE "${PROJECT_SOURCE_DIR}/src/AudioCore/src")
add_executable(CoreDevice.Tests "${PROJECT_SOURCE_DIR}/tests/CoreDeviceTests.cpp")
target_link_libraries(CoreDevice.Tests PRIVATE AudioCoreImpl)
add_dependencies(CoreDevice.Tests CoreDevice.TestHost)
add_test(NAME CoreDevice.Tests COMMAND CoreDevice.Tests $<TARGET_FILE:CoreDevice.TestHost>)
set_tests_properties(CoreDevice.Tests PROPERTIES TIMEOUT 30)
if(TARGET CoreDevice.Runtime)
  add_test(NAME CoreDevice.ConnectionFailure COMMAND CoreDevice.Tests "${AB_COREDEVICE_RUNTIME}/AudioBridge.CoreDeviceHost.exe" --missing-device)
  set_tests_properties(CoreDevice.ConnectionFailure PROPERTIES TIMEOUT 20)
endif()
