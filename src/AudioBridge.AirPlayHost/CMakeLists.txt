# Pinned upstream receiver host; audio-only customization is recorded in vendor/BRIDGE.md.
add_library(AudioBridge.DnsSd SHARED "${AIRPLAY}/WirelessHost/DnsSdShim.cpp" "${AIRPLAY}/WirelessHost/DnsSdShim.def")
target_compile_definitions(AudioBridge.DnsSd PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(AudioBridge.DnsSd PRIVATE dnsapi iphlpapi ws2_32)
set_target_properties(AudioBridge.DnsSd PROPERTIES PREFIX "" OUTPUT_NAME dnssd RUNTIME_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/Wireless")
add_executable(AudioBridge.AirPlayHost "${AIRPLAY}/WirelessHost/WirelessHost.cpp" "${AIRPLAY}/WirelessHost/DlnaRenderer.cpp")
target_compile_definitions(AudioBridge.AirPlayHost PRIVATE ZL_AIRPLAY_AUDIO_ONLY=1)
target_link_libraries(AudioBridge.AirPlayHost PRIVATE bcrypt ws2_32 winhttp)
set_target_properties(AudioBridge.AirPlayHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/Wireless")
add_dependencies(AudioBridge.AirPlayHost AudioBridge.DnsSd)
foreach(runtime airplay2dll.dll avcodec-58.dll avutil-56.dll swresample-3.dll swscale-5.dll)
  configure_file("${AIRPLAY}/airplay-server/bin/x64/${runtime}" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/Wireless/${runtime}" COPYONLY)
endforeach()
foreach(runtime msvcp140.dll vcruntime140.dll vcruntime140_1.dll)
  configure_file("${AIRPLAY}/msvc-runtime/${runtime}" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/Wireless/${runtime}" COPYONLY)
endforeach()
